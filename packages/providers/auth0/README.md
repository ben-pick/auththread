# @auththread/auth0

Auth0 deep adapter: Universal Login, Actions, and Log Streams.

## Install

```sh
pnpm add @auththread/auth0
```

## Usage

```ts
import { auth0Adapter } from "@auththread/auth0";
import { auth0 } from "@auththread/auth0/browser";
import { withTrace } from "@auththread/auth0/actions";
```
